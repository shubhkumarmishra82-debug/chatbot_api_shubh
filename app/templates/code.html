Call the API from any website (JavaScript)

Drop this into your site's JS. Replace YOUR_GMS_API_KEY with your real key from /api.

async function askGMS(message) {
  const res = await fetch('https://chatbot-api-shubh.vercel.app/v1/chat/completions', {
    method: 'POST',
    headers: {
      'Content-Type': 'application/json',
      'Authorization': 'Bearer YOUR_GMS_API_KEY'
    },
    body: JSON.stringify({
      messages: [{ role: 'user', content: message }],
      conversation_id: 'visitor-123'   // optional: keeps memory per user
    })
  });
  const data = await res.json();
  return data.choices[0].message.content;
}

// usage:
askGMS("hello").then(reply => console.log(reply));

Set conversation_id to something unique per visitor (e.g. a session ID) so the bot remembers context across their messages.

Telegram bot (Python)

Requires: pip install python-telegram-bot requests. Get a bot token from @BotFather.

import requests
from telegram import Update
from telegram.ext import Application, MessageHandler, ContextTypes, filters

GMS_API_KEY = "YOUR_GMS_API_KEY"
GMS_URL = "https://chatbot-api-shubh.vercel.app/v1/chat/completions"
TELEGRAM_TOKEN = "YOUR_TELEGRAM_BOT_TOKEN"

async def handle_message(update: Update, context: ContextTypes.DEFAULT_TYPE):
    user_msg = update.message.text
    chat_id = str(update.effective_chat.id)

    resp = requests.post(
        GMS_URL,
        headers={"Authorization": f"Bearer {GMS_API_KEY}"},
        json={
            "messages": [{"role": "user", "content": user_msg}],
            "conversation_id": f"telegram-{chat_id}"
        },
    )
    reply = resp.json()["choices"][0]["message"]["content"]
    await update.message.reply_text(reply)

app = Application.builder().token(TELEGRAM_TOKEN).build()
app.add_handler(MessageHandler(filters.TEXT & ~filters.COMMAND, handle_message))
app.run_polling()

Run this on any always-on machine (a small VPS, your PC, etc) with: python bot.py

Discord bot (Python)

Requires: pip install discord.py requests. Create a bot + get its token at discord.com/developers.

import discord
import requests

GMS_API_KEY = "YOUR_GMS_API_KEY"
GMS_URL = "https://chatbot-api-shubh.vercel.app/v1/chat/completions"
DISCORD_TOKEN = "YOUR_DISCORD_BOT_TOKEN"

intents = discord.Intents.default()
intents.message_content = True
client = discord.Client(intents=intents)

@client.event
async def on_message(message):
    if message.author == client.user:
        return

    resp = requests.post(
        GMS_URL,
        headers={"Authorization": f"Bearer {GMS_API_KEY}"},
        json={
            "messages": [{"role": "user", "content": message.content}],
            "conversation_id": f"discord-{message.channel.id}"
        },
    )
    reply = resp.json()["choices"][0]["message"]["content"]
    await message.channel.send(reply)

client.run(DISCORD_TOKEN)

In the Discord Developer Portal, enable "Message Content Intent" under Bot settings, or on_message won't receive text.
